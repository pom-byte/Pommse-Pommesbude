import discord
from discord.ext import commands

Aktiviert die benötigten Berechtigungen
intents = discord.Intents.default()
intents.message_content = True  # Erlaubt das Lesen von Nachrichten

bot = commands.Bot(command_prefix="!", intents=intents)

Event: Wird ausgeführt, wenn der Bot online geht
@bot.event
async def on_ready():
    print(f'Eingeloggt als {bot.user}')

Befehl: Reagiert auf "!ping"
@bot.command()
async def ping(ctx):
    await ctx.send('Pong! 🏓')

Bot starten
